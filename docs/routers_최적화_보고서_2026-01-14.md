   ```python
           conn.rollback()
           raise
       finally:
           cursor.close()
           conn.close()
   
   # After

   async def signup(request: SignupRequest, cursor=None, conn=None):

   ```

3. **camelCase 응답 형식 적용**
   ```python

       "memberId": 1,
   ```

   - Before: `router = APIRouter()`

5. **Import 정리**
   - 추가: `from utils.common import handle_db_transaction`
   - 추가: `from models.responses import ...`
   - 추가: `import re` (상단으로 이동)

**최적화 효과**:

### 3. 에러 처리 자동화

**Before (수동 에러 처리)**:
```python
try:
    # 비즈니스 로직
    conn.commit()
    return result
except mysql.connector.Error as err:
    conn.rollback()
    raise HTTPException(status_code=500, detail=f"데이터베이스 오류: {str(err)}")
except Exception as e:
    conn.rollback()
    raise HTTPException(status_code=500, detail=str(e))
finally:
    cursor.close()
    conn.close()
```

**After (자동 에러 처리)**:
```python
@handle_db_transaction
async def endpoint(..., cursor=None, conn=None):
    # 비즈니스 로직만
    return result
    
# 데코레이터가 자동으로:
# - commit() 실행
# - 에러 시 rollback()
# - 연결 종료 (cursor.close(), conn.close())
# - HTTPException 발생
```

```python
def validate_post_exists(cursor, post_id: int) -> dict:
    """게시물 존재 확인"""
    cursor.execute("SELECT * FROM post WHERE post_id = %s AND deleted_at IS NULL", (post_id,))
    post = cursor.fetchone()
    if not post:
        raise HTTPException(status_code=404, detail="게시물을 찾을 수 없습니다")
    return post

# 2. check_ownership
def check_ownership(current_user_id: int, resource_user_id: int):
    """본인 리소스인지 확인"""
```
